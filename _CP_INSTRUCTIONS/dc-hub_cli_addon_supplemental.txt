# ──────────────────────────────────────────────────────────────────────────
# 3. CLI TEMPLATE
# ──────────────────────────────────────────────────────────────────────────

<<-EOT
interface Vlan{{cli_vlan_corp}}
 vrf forwarding 10
 ip pim sparse-mode
 ip pim dr-priority {{cli_pim_dr_priority}}
 bfd template bfd1
!
ip access-list standard ACL_PIM2
 10 permit 224.0.0.0 15.255.255.255
!
ip pim vrf 10 rp-address ${var.cli_addon.pim.rp_address} ACL_PIM2
!
interface ${var.site_cfg.transports["private1"].interface}
 ${var.cli_addon.private_interface.speed_config}
!
bfd-template single-hop bfd1
 interval min-tx 500 min-rx 500 multiplier 3
!
interface Vlan{{cli_vlan_infra}}
 bfd template bfd1
!
router bgp ${var.cli_addon.bgp.as_number}
 address-family ipv4 vrf 10
  neighbor {{cli_bgp_neighbor_infra}} fall-over bfd
  neighbor {{cli_bgp_neighbor_corp}} fall-over bfd
!
ip access-list extended BUF-FILTER
 10 permit udp host ${var.cli_addon.buf_filter.host_a} host ${var.cli_addon.buf_filter.host_b}
 20 permit tcp host ${var.cli_addon.buf_filter.host_a} host ${var.cli_addon.buf_filter.host_b}
 30 permit udp host ${var.cli_addon.buf_filter.host_b} host ${var.cli_addon.buf_filter.host_a}
 40 permit tcp  host ${var.cli_addon.buf_filter.host_b} host ${var.cli_addon.buf_filter.host_a}
!
pnp startup-vlan ${var.cli_addon.pnp_startup_vlan}
!
EOT


# ──────────────────────────────────────────────────────────────────────────
# 4. HARDCODED VALUES — rationale
# ──────────────────────────────────────────────────────────────────────────
#
# ACL_PIM2                      — org-wide ACL name
# 224.0.0.0 15.255.255.255      — full multicast range; standard
# vrf forwarding 10             — always VPN 10
# ip pim sparse-mode            — always PIM-SM
# bfd-template single-hop bfd1  — name and type fixed org-wide
# interval 500/500/3            — org-wide BFD timers; promote to YAML only
#                                  if site-level tuning is ever needed
# address-family ipv4 vrf 10    — always VPN 10
# fall-over bfd                 — feature behaviour, not configurable
